function result = Z(A, A_pseudo_inverse)
    result = eye(size(A_pseudo_inverse, 1)) - A_pseudo_inverse * A;
end

function result = GrevillePseudoInversion(A)
    is_swap = 0;
    if(size(A, 1) > size(A, 2))
        is_swap = 1;
        A = A';
    end
    
    current_vector = A(1, :)';
    A_pseudo_inverse = 0;
    vector_scalar = current_vector' * current_vector;
    
    if(vector_scalar == 0)
        A_pseudo_inverse = current_vector;
    else
        A_pseudo_inverse = current_vector / vector_scalar;
    end
    
    A_i = current_vector';
    
    for i = 2 : size(A, 1)
        current_vector = A(i, :)';
        Z_A = Z(A_i, A_pseudo_inverse);
        A_i = [A_i; current_vector'];
        
        denom_Z = current_vector' * Z_A * current_vector;
        
        if(denom_Z > 0)
            A_pseudo_inverse = [(A_pseudo_inverse - (Z_A * current_vector * current_vector' * A_pseudo_inverse) / denom_Z), (Z_A * current_vector) / denom_Z];
        else
            R_A = A_pseudo_inverse * A_pseudo_inverse';
            denom_R = 1 + current_vector' * R_A * current_vector;
            A_pseudo_inverse = [(A_pseudo_inverse - (R_A * current_vector * current_vector' * A_pseudo_inverse) / denom_R), (R_A * current_vector) / denom_R];
        end
    end
    
    if (is_swap)
        A_pseudo_inverse = A_pseudo_inverse';
    end
    
    result = A_pseudo_inverse;
end

function result = J(A_current, A_next)
    result = max(max((A_current - A_next) .^ 2));
end

function result = MoorePenrosePseudoInversion(A)
    is_swap = 0;
    if(size(A, 1) > size(A, 2))
        is_swap = 1;
        A = A';
    end
    
    CONST_E = 1e-8;
    CONST_INF = 1e9;
    delta = 10.0;
    
    A_pseudo_inverse_current = CONST_INF * ones(size(A))';
    A_pseudo_inverse_next = -CONST_INF * ones(size(A))';
    
    while(J(A_pseudo_inverse_current, A_pseudo_inverse_next) > CONST_E)
        A_pseudo_inverse_current = A_pseudo_inverse_next;
        A_pseudo_inverse_next = A' * inv(A * A' + (delta) * eye(size(A, 1)));
        delta = delta / 2.0;
    end
    
    if(is_swap)
        A_pseudo_inverse_next = A_pseudo_inverse_next';
    end
    
    result = A_pseudo_inverse_next;
end
